cmake_minimum_required(VERSION 3.24)
project(untitled_engine VERSION 0.1.0 LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/CompilerOptions.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Dependencies.cmake")

option(ENGINE_BUILD_SANDBOX "Build sandbox app" ON)
option(ENGINE_BUILD_ERA_SANDBOX "Build era sandbox app" ON)
option(ENGINE_BUILD_TESTS "Build tests" ON)

add_subdirectory(engine/foundation)
add_subdirectory(engine/world)
add_subdirectory(engine/runtime)
add_subdirectory(engine/modules/save)
add_subdirectory(engine/modules/data)

add_subdirectory(engine/core)
add_subdirectory(engine/renderer/vulkan)
add_subdirectory(engine/integration)
add_subdirectory(engine/assets)
add_subdirectory(engine/ecs)
add_subdirectory(engine/physics)
add_subdirectory(engine/audio)
add_subdirectory(engine/vfx)

if(ENGINE_BUILD_SANDBOX)
    add_subdirectory(apps/sandbox)
endif()

if(ENGINE_BUILD_ERA_SANDBOX)
    add_subdirectory(apps/era_sandbox)
endif()

if(ENGINE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
    if(ENGINE_BUILD_SANDBOX)
        add_test(NAME smoke_test_sandbox COMMAND sandbox --smoke-test)
    endif()
    if(ENGINE_BUILD_ERA_SANDBOX)
        add_test(NAME smoke_test_era_sandbox COMMAND era_sandbox --smoke-test)
    endif()
endif()
